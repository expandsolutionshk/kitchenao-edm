$ao_t = function ( $v ) { $x = ao_en_seo_get( 't' ); return $x ? $x : $v; };
$ao_d = function ( $v ) { $x = ao_en_seo_get( 'd' ); return $x ? $x : $v; };

add_filter( 'rank_math/frontend/title', $ao_t, 99 );
add_filter( 'rank_math/frontend/description', $ao_d, 99 );
add_filter( 'rank_math/opengraph/facebook/og_title', $ao_t, 99 );
add_filter( 'rank_math/opengraph/facebook/og_description', $ao_d, 99 );
add_filter( 'rank_math/opengraph/twitter/title', $ao_t, 99 );
add_filter( 'rank_math/opengraph/twitter/description', $ao_d, 99 );
add_filter( 'rank_math/opengraph/facebook/og_locale', function ( $v ) {
	global $TRP_LANGUAGE;
	return ( ! empty( $TRP_LANGUAGE ) && strpos( $TRP_LANGUAGE, 'en' ) === 0 ) ? 'en_GB' : $v;
}, 99 );
