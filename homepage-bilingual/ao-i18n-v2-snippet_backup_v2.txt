/**
 * Kitchen AO – 中英文版通用工具 v2（只需上載一次）
 * 1) 每篇頁面／文章／產品的編輯畫面多一個「English SEO」方格，填英文 Title 及 Description，只在 /en/ 生效
 *    （未填的頁面會用舊 EN SEO snippet 的 $map 或 Rank Math 中文設定）
 *    產品分類（product_cat）及文章分類（category）的編輯畫面亦有同樣兩欄
 * 2) WPBakery 任何元件或 Row 的「Extra class name」填 ao-en ＝只在英文版顯示；填 ao-zh ＝只在中文版顯示
 * 3) 沒有用 WPBakery 的頁面（傳統編輯器／區塊編輯器）：用 [ao_zh]中文內容[/ao_zh] 及 [ao_en]English[/ao_en] 包住內容
 * WPCode：PHP Snippet，Auto Insert › Run Everywhere
 */

function ao_is_en() {
	global $TRP_LANGUAGE;
	return ! empty( $TRP_LANGUAGE ) && strpos( $TRP_LANGUAGE, 'en' ) === 0;
}

/* ---------- 1. English SEO 方格 ---------- */
add_action( 'add_meta_boxes', function () {
	foreach ( array( 'page', 'post', 'product' ) as $pt ) {
		add_meta_box( 'ao_en_seo', 'English SEO（只在 /en/ 生效）', 'ao_en_seo_box', $pt, 'normal', 'high' );
	}
} );

function ao_en_seo_box( $post ) {
	wp_nonce_field( 'ao_en_seo_save', 'ao_en_seo_nonce' );
	$t = get_post_meta( $post->ID, '_ao_en_title', true );
	$d = get_post_meta( $post->ID, '_ao_en_desc', true );
	echo '<p><label><b>English Title</b>（建議 60 字元以內）<br><input type="text" name="ao_en_title" value="' . esc_attr( $t ) . '" style="width:100%"></label></p>';
	echo '<p><label><b>English Meta Description</b>（建議 160 字元以內）<br><textarea name="ao_en_desc" rows="3" style="width:100%">' . esc_textarea( $d ) . '</textarea></label></p>';
	echo '<p style="color:#666">留空＝英文版沿用中文設定。</p>';
}

add_action( 'save_post', function ( $post_id ) {
	$nonce = filter_input( INPUT_POST, 'ao_en_seo_nonce' );
	if ( ! $nonce || ! wp_verify_nonce( $nonce, 'ao_en_seo_save' ) ) return;
	if ( defined( 'DOING_AUTOSAVE' ) && DOING_AUTOSAVE ) return;
	if ( ! current_user_can( 'edit_post', $post_id ) ) return;
	$t = (string) filter_input( INPUT_POST, 'ao_en_title' );
	$d = (string) filter_input( INPUT_POST, 'ao_en_desc' );
	update_post_meta( $post_id, '_ao_en_title', sanitize_text_field( $t ) );
	update_post_meta( $post_id, '_ao_en_desc', sanitize_textarea_field( $d ) );
} );

/* 分類頁（產品分類、文章分類）English SEO 欄位 */
foreach ( array( 'product_cat', 'category' ) as $ao_tax ) {
	add_action( $ao_tax . '_edit_form_fields', function ( $term ) {
		$t = get_term_meta( $term->term_id, '_ao_en_title', true );
		$d = get_term_meta( $term->term_id, '_ao_en_desc', true );
		wp_nonce_field( 'ao_en_seo_term', 'ao_en_seo_term_nonce' );
		echo '<tr class="form-field"><th scope="row">English Title（只在 /en/ 生效）</th><td><input type="text" name="ao_en_title" value="' . esc_attr( $t ) . '"></td></tr>';
		echo '<tr class="form-field"><th scope="row">English Meta Description</th><td><textarea name="ao_en_desc" rows="3">' . esc_textarea( $d ) . '</textarea></td></tr>';
	} );
	add_action( 'edited_' . $ao_tax, function ( $term_id ) {
		$nonce = filter_input( INPUT_POST, 'ao_en_seo_term_nonce' );
		if ( ! $nonce || ! wp_verify_nonce( $nonce, 'ao_en_seo_term' ) ) return;
		if ( ! current_user_can( 'manage_categories' ) && ! current_user_can( 'manage_product_terms' ) ) return;
		update_term_meta( $term_id, '_ao_en_title', sanitize_text_field( (string) filter_input( INPUT_POST, 'ao_en_title' ) ) );
		update_term_meta( $term_id, '_ao_en_desc', sanitize_textarea_field( (string) filter_input( INPUT_POST, 'ao_en_desc' ) ) );
	} );
}

function ao_en_meta( $key ) {
	if ( ! ao_is_en() ) return null;
	$mk = $key === 't' ? '_ao_en_title' : '_ao_en_desc';
	if ( is_category() || is_tax( 'product_cat' ) ) {
		$v = get_term_meta( get_queried_object_id(), $mk, true );
		return $v !== '' ? $v : null;
	}
	if ( is_front_page() ) $id = (int) get_option( 'page_on_front' );
	elseif ( is_home() ) $id = (int) get_option( 'page_for_posts' );            // Blog 列表頁
	elseif ( function_exists( 'is_shop' ) && is_shop() ) $id = (int) wc_get_page_id( 'shop' ); // 商店頁
	elseif ( is_singular() ) $id = get_queried_object_id();
	else $id = 0;
	if ( ! $id ) return null;
	$v = get_post_meta( $id, $mk, true );
	return $v !== '' ? $v : null;
}

$ao_v2_t = function ( $v ) { $x = ao_en_meta( 't' ); return $x ? $x : $v; };
$ao_v2_d = function ( $v ) { $x = ao_en_meta( 'd' ); return $x ? $x : $v; };
// priority 100：比舊 EN SEO snippet（99）遲執行，方格有填就以方格為準
add_filter( 'rank_math/frontend/title', $ao_v2_t, 100 );
add_filter( 'rank_math/frontend/description', $ao_v2_d, 100 );
add_filter( 'rank_math/opengraph/facebook/og_title', $ao_v2_t, 100 );
add_filter( 'rank_math/opengraph/facebook/og_description', $ao_v2_d, 100 );
add_filter( 'rank_math/opengraph/twitter/title', $ao_v2_t, 100 );
add_filter( 'rank_math/opengraph/twitter/description', $ao_v2_d, 100 );

/* ---------- 2. WPBakery：按語言顯示元件 ---------- */
add_filter( 'vc_shortcode_output', function ( $output, $shortcode = null, $atts = array() ) {
	if ( is_admin() || empty( $atts['el_class'] ) ) return $output;
	$cls = ' ' . $atts['el_class'] . ' ';
	$en  = ao_is_en();
	if ( ! $en && strpos( $cls, ' ao-en ' ) !== false ) return '';
	if ( $en && strpos( $cls, ' ao-zh ' ) !== false ) return '';
	return $output;
}, 10, 3 );

/* ---------- 3. 傳統／區塊編輯器：[ao_zh] [ao_en] 短碼 ---------- */
add_shortcode( 'ao_zh', function ( $atts, $content = '' ) { return ao_is_en() ? '' : do_shortcode( $content ); } );
add_shortcode( 'ao_en', function ( $atts, $content = '' ) { return ao_is_en() ? do_shortcode( $content ) : ''; } );
