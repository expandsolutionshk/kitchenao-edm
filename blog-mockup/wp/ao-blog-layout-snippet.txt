 * - products：產品 ID（逗號分隔），沒有圖片的產品自動不顯示
 * - related：延伸閱讀文章 ID；留空＝自動揀同分類、有特色圖片的 3 篇

$GLOBALS['ao_bl_atts'] = null;

/* 1) Shortcode：只記下參數，不輸出任何內容 */
add_shortcode( 'ao_blog', function ( $atts ) {
	$GLOBALS['ao_bl_atts'] = shortcode_atts( array(
		'products'  => '',
		'related'   => '',
		'more_url'  => '',
		'more_text' => '查看全部套餐',
		'cta_title' => '想訂到會？',
		'cta_text'  => '告訴我們日子、人數及飲食要求，客服為你配搭餐單。',
		'cta_url'   => 'https://wa.me/85269011987',
	), $atts, 'ao_blog' );
	return '';
} );

function ao_bl_on() {
	if ( is_admin() || ! is_singular( 'post' ) ) { return false; }
	$p = get_post();
	return $p && has_shortcode( $p->post_content, 'ao_blog' );
}

/* 2) body class（用來放寬主題 980px 的內容闊度） */
add_filter( 'body_class', function ( $c ) {
	if ( ao_bl_on() ) { $c[] = 'ao-bl-on'; }
	return $c;
} );

/* 3) 產品、延伸閱讀資料 */
function ao_bl_products( $ids ) {
	$out = array();
	if ( ! function_exists( 'wc_get_product' ) ) { return $out; }
	foreach ( array_filter( array_map( 'absint', explode( ',', $ids ) ) ) as $id ) {
		$p = wc_get_product( $id );
		if ( ! $p || 'publish' !== $p->get_status() ) { continue; }
		$img = $p->get_image_id() ? wp_get_attachment_image_url( $p->get_image_id(), 'woocommerce_thumbnail' ) : '';
		if ( ! $img ) { continue; }
		$out[] = array(
			'name'  => $p->get_name(),
			'url'   => get_permalink( $id ),
			'img'   => $img,
			'price' => wp_strip_all_tags( wc_price( (float) $p->get_price(), array( 'decimals' => 0 ) ) ),
		);
	}
	return $out;
}
function ao_bl_related( $ids, $self ) {
	$list = array_filter( array_map( 'absint', explode( ',', $ids ) ) );
	if ( ! $list ) {
		$cats = wp_get_post_categories( $self );
		$q    = get_posts( array( 'post_type' => 'post', 'posts_per_page' => 6, 'category__in' => $cats, 'post__not_in' => array( $self ), 'fields' => 'ids' ) );
		$list = $q;
	}
	$out = array();
	foreach ( $list as $id ) {
		if ( (int) $id === (int) $self || 'publish' !== get_post_status( $id ) ) { continue; }
		$img = get_the_post_thumbnail_url( $id, 'medium' );
		if ( ! $img ) { continue; }
		$out[] = array( 'title' => get_the_title( $id ), 'url' => get_permalink( $id ), 'img' => $img );
		if ( count( $out ) >= 3 ) { break; }
	}
	return $out;
}

/* 4) 包住文章：左邊文章＋文末區，右邊側欄 */
add_filter( 'the_content', function ( $html ) {
	static $done = false;
	if ( $done || ! ao_bl_on() || ! in_the_loop() || ! is_main_query() || empty( $GLOBALS['ao_bl_atts'] ) ) { return $html; }
	$done = true;
	$a    = $GLOBALS['ao_bl_atts'];
	$pid  = get_the_ID();
	$prod = ao_bl_products( $a['products'] );
	$rel  = ao_bl_related( $a['related'], $pid );
	$mins = max( 1, (int) ceil( mb_strlen( wp_strip_all_tags( $html ) ) / 450 ) );
	$more = $a['more_url'] ? ( 0 === strpos( $a['more_url'], 'http' ) ? $a['more_url'] : home_url( $a['more_url'] ) ) : '';
	$up   = (int) get_post_meta( $pid, '_ao_vote_up', true );
	$dn   = (int) get_post_meta( $pid, '_ao_vote_dn', true );

	/* 側欄產品 */
	$side = '';
	foreach ( $prod as $p ) {
		$side .= '<a class="ao-pi" href="' . esc_url( $p['url'] ) . '"><img src="' . esc_url( $p['img'] ) . '" alt="" loading="lazy"><div><b>' . esc_html( $p['name'] ) . '</b><span>' . esc_html( $p['price'] ) . '</span></div></a>';
	}
	/* 文末產品格 */
	$grid = '';
	foreach ( $prod as $p ) {
		$grid .= '<a class="ao-pc" href="' . esc_url( $p['url'] ) . '"><img src="' . esc_url( $p['img'] ) . '" alt="' . esc_attr( $p['name'] ) . '" loading="lazy"><div><b>' . esc_html( $p['name'] ) . '</b><span>' . esc_html( $p['price'] ) . '</span><em>查看套餐</em></div></a>';
	}
	/* 延伸閱讀 */
	$rg = '';
	foreach ( $rel as $r ) {
		$rg .= '<a class="ao-rc" href="' . esc_url( $r['url'] ) . '"><img src="' . esc_url( $r['img'] ) . '" alt="" loading="lazy"><b>' . esc_html( $r['title'] ) . '</b></a>';
	}
	/* Tag */
	$tg = '';
	foreach ( (array) get_the_tags( $pid ) as $t ) {
		if ( $t ) { $tg .= '<a href="' . esc_url( get_tag_link( $t ) ) . '">' . esc_html( $t->name ) . '</a>'; }
	}

	$end  = '<div class="ao-end">';
	$end .= '<div class="ao-vote" data-id="' . $pid . '"><button type="button" class="cp" data-copy>🔗 複製文章連結</button><span class="q">呢篇文章有冇幫到你？</span><button type="button" class="b" data-v="up">👍 有用</button><button type="button" class="b" data-v="dn">👎 一般</button>';
	$end .= '<div class="ao-vbar"><i style="width:' . ( $up + $dn ? round( $up / ( $up + $dn ) * 100 ) : 0 ) . '%"></i></div><div class="ao-vtxt" data-up="' . $up . '" data-dn="' . $dn . '"></div></div>';
	if ( $tg )   { $end .= '<div class="ao-sec"><h3>此文章相關 TAG</h3><div class="ao-tags">' . $tg . '</div></div>'; }
	if ( $grid ) { $end .= '<div class="ao-sec"><h3>推介產品</h3><div class="ao-pg">' . $grid . '</div></div>'; }
	if ( $rg )   { $end .= '<div class="ao-sec"><h3>📖 延伸閱讀</h3><div class="ao-rg">' . $rg . '</div></div>'; }
	$end .= '</div>';

	$aside  = '<aside class="ao-bl-aside">';
	$aside .= '<div class="ao-meta"><span>作者</span><b>Kitchen AO 編輯部</b><span>更新</span><b>' . esc_html( get_the_modified_date( 'Y-m-d', $pid ) ) . '</b><span>閱讀</span><b>約 ' . $mins . ' 分鐘</b></div>';
	$aside .= '<div class="ao-sc toc"><h4>目錄</h4><div class="ao-prog"><i></i></div><ol class="ao-toc2"></ol></div>';
	$aside .= '<div class="ao-bshare"><b>分享</b><a href="#" class="cp" data-copy title="複製連結"><i class="fas fa-link"></i><em>複製</em></a><a href="#" class="wa" target="_blank" rel="noopener" title="分享到 WhatsApp"><i class="fab fa-whatsapp"></i><em>WhatsApp</em></a><a href="#" class="fb" target="_blank" rel="noopener" title="分享到 Facebook"><i class="fab fa-facebook-f"></i><em>Facebook</em></a><a href="#" class="ig" title="分享到 Instagram"><i class="fab fa-instagram"></i><em>IG</em></a></div>';
	if ( $side ) {
		$aside .= '<div class="ao-sc"><h4>推介產品</h4><div class="ao-pl">' . $side . '</div>' . ( $more ? '<a class="ao-more" href="' . esc_url( $more ) . '">' . esc_html( $a['more_text'] ) . ' →</a>' : '' ) . '</div>';
	}
	$aside .= '<div class="ao-cta2"><b>' . esc_html( $a['cta_title'] ) . '</b><p>' . esc_html( $a['cta_text'] ) . '</p><a href="' . esc_url( $a['cta_url'] ) . '" target="_blank" rel="noopener">WhatsApp 查詢</a></div>';
	$aside .= '</aside>';

	return '<div class="ao-bl"><div class="ao-bl-main" id="ao-art">' . $html . $end . '</div>' . $aside . '</div>';
}, 999 );

/* 5) 投票（REST，不使用表單變數） */
add_action( 'rest_api_init', function () {
	register_rest_route( 'ao/v1', '/vote', array(
		'methods'             => 'POST',
		'permission_callback' => '__return_true',
		'callback'            => function ( WP_REST_Request $r ) {
			$id = absint( $r->get_param( 'id' ) );
			$v  = $r->get_param( 'v' ) === 'up' ? '_ao_vote_up' : '_ao_vote_dn';
			if ( ! $id || 'post' !== get_post_type( $id ) ) { return new WP_Error( 'bad', 'bad', array( 'status' => 400 ) ); }
			update_post_meta( $id, $v, (int) get_post_meta( $id, $v, true ) + 1 );
			return array( 'up' => (int) get_post_meta( $id, '_ao_vote_up', true ), 'dn' => (int) get_post_meta( $id, '_ao_vote_dn', true ) );
		},
	) );
} );
